package ru.rtkit.lukomorye.core;

@Service
public class PluginOrchestrator {

  private final PluginRegistry<ExtensionPoint, String> registry;
  private final KafkaTemplate<String, Event> kafka;

  @Transactional
  public Result dispatch(Command cmd) {
    var plugin = registry.getPluginFor(cmd.type())
        .orElseThrow(PluginNotFoundException::new);

    var result = plugin.execute(cmd);
    kafka.send("domain.events", result.toEvent());
    return result;
  }
}
